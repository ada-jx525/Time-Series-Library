# Server core comparison commands: baseline checkpoint first, then base-aware memory.
# Run from repository root.
#
# Important:
#   1. For each base model and pred_len, run the baseline command first.
#   2. The memory command loads the matching trained baseline via --wm_base_checkpoint.
#   3. Do not run the memory command before its baseline checkpoint exists.
#
# Shared formal setting:
#   configs/benchmark/etth1_common.args       -> ETTh1, seq_len=96, train_epochs=150, patience=10
#   configs/benchmark/branchworld_common.args -> BranchWorld memory defaults, wm_freeze_base=1
#
# To use another GPU, change CUDA_VISIBLE_DEVICES and --gpu in configs/benchmark/etth1_common.args.

# pred_len = 96
# =========================
